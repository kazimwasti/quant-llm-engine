# ---------------------------------------------------------------------------
# Arm 2 first: regenerate the manager dataset (overwrites any stale copy),
# then train the manager adapter. Fast, so it is banked before the long run.
# ---------------------------------------------------------------------------
!python scripts/generate_manager_dataset.py --output data/manager_train.jsonl

import json
rows = [json.loads(line) for line in open('data/manager_train.jsonl', encoding='utf-8')]
scored = sum(1 for r in rows if r['meta']['n_scored_grid_points'] > 0)
assert len(rows) >= 200 and scored >= 50, (
    f'unexpected dataset shape: {len(rows)} rows, {scored} scored — '
    'check that the remote repo has the final generate_manager_dataset.py'
)
print(f'manager dataset OK: {len(rows)} rows, {scored} scored')

!python scripts/train_manager_lora.py \
    --data data/manager_train.jsonl \
    --output-dir models/manager_adapter \
    --epochs 4

# ---------------------------------------------------------------------------
# Phase 2 — decision sub-agent. Regenerate the training set from a fresh
# matched 5m/hourly cloud pair (the committed JSONL predates the current
# prompt format), then train. The trainer R-weights its loss: winners teach
# more, gated-candidate HOLDs weigh full, background HOLDs weigh 0.25.
# ---------------------------------------------------------------------------
if TRAIN_DECISION_LLM:
    !python scripts/prepare_cloud_data.py --symbol MNQ=F --test-days 50
    !python scripts/dataset_generator.py \
        --input data/5m_cloud.csv --hourly-input data/hourly_cloud.csv \
        --output data/jev_train_cloud.jsonl
    import json
    rows = [json.loads(line) for line in open('data/jev_train_cloud.jsonl', encoding='utf-8')]
    with_r = sum(1 for r in rows if r.get('meta', {}).get('realized_r') is not None)
    assert with_r >= 20, f'expected realized_r barrier outcomes, got {with_r}'
    print(f'decision dataset OK: {len(rows)} rows, {with_r} with realized_r outcomes')
    !python scripts/train_lora.py \
        --data data/jev_train_cloud.jsonl \
        --output-dir models/qwen_adapter \
        --epochs 3 \
        --r-weight-scale 1.0 --hold-weight 0.25
